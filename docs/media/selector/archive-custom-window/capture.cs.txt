var scene = UnityEditor.SceneManagement.EditorSceneManager.NewPreviewScene();
var go = new UnityEngine.GameObject("Loadout preview");
UnityEngine.SceneManagement.SceneManager.MoveGameObjectToScene(go, scene);
var loadout = go.AddComponent<Aspid.FastTools.Samples.SerializeReferences.Loadout>();
var so = new UnityEditor.SerializedObject(loadout);
var primary = so.FindProperty("_primary");
primary.managedReferenceValue = new Aspid.FastTools.Samples.SerializeReferences.Pistol();
so.ApplyModifiedPropertiesWithoutUndo();
primary.FindPropertyRelative("_damage").intValue = 37;
primary.isExpanded = true;
so.ApplyModifiedPropertiesWithoutUndo();
var w = UnityEngine.ScriptableObject.CreateInstance<UnityEditor.EditorWindow>();
w.titleContent = new UnityEngine.GUIContent("SerializeReference Selector");
var root = w.rootVisualElement;
root.style.backgroundColor = new UnityEngine.Color(0.22f, 0.22f, 0.22f);
root.style.paddingLeft = 16;
root.style.paddingRight = 16;
root.style.paddingTop = 16;
root.style.paddingBottom = 16;
var title = new UnityEngine.UIElements.Label("Loadout");
title.style.fontSize = 20;
title.style.unityFontStyleAndWeight = UnityEngine.FontStyle.Bold;
title.style.marginBottom = 16;
root.Add(title);
root.Add(new UnityEditor.UIElements.PropertyField(primary));
root.Add(new UnityEditor.UIElements.PropertyField(so.FindProperty("_sidearms")));
UnityEditor.UIElements.BindingExtensions.Bind(root, so);
root.style.width = 640;
root.style.height = 350;
root.style.transformOrigin = new UnityEngine.UIElements.TransformOrigin(0, 0, 0);
root.style.scale = new UnityEngine.UIElements.Scale(new UnityEngine.Vector3(1.5f,1.5f,1));
w.ShowUtility();
w.minSize = new UnityEngine.Vector2(960, 525);
w.maxSize = new UnityEngine.Vector2(960, 525);
w.position = new UnityEngine.Rect(120,120,960,525);
w.Focus();
return new { window=w.GetInstanceID(), objectId=go.GetInstanceID(), scene=scene.handle, target=loadout.GetInstanceID() };
