using System;
using System.Linq;
using System.Reflection;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine.SceneManagement;
public static class PrepareEnumCapture
{
 public static object Main(string theme)
 {
  if(EditorApplication.isPlaying || SceneManager.GetActiveScene().isDirty)
   throw new Exception("Stop Play Mode and use a clean scene");
  var version=UnityEditor.PackageManager.PackageInfo.FindForAssetPath("Packages/tech.aspid.fasttools/package.json").version;
  EditorSceneManager.OpenScene("Assets/Samples/Aspid.FastTools/"+version+"/EnumValues/Scenes/EnumValues.unity");
  var type=AppDomain.CurrentDomain.GetAssemblies().First(a=>a.GetName().Name=="Aspid.FastTools.Editor")
   .GetType("Aspid.FastTools.Editors.Internal.SampleThemePreview");
  var method=type.GetMethod("SetMode",BindingFlags.Static|BindingFlags.NonPublic);
  method.Invoke(null,new[]{Enum.Parse(method.GetParameters()[0].ParameterType,theme)});
  return new{theme,scene=SceneManager.GetActiveScene().path};
 }
}
