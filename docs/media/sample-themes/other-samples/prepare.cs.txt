using System;
using System.IO;
using System.Linq;
using System.Reflection;
using UnityEngine;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine.SceneManagement;
public static class PrepareSamples {
 const BindingFlags Flags=BindingFlags.NonPublic|BindingFlags.Public|BindingFlags.Static|BindingFlags.Instance;
 public static object Main(string sample) {
  var scene=SceneManager.GetActiveScene();
  if(scene.isDirty||EditorApplication.isPlaying)throw new Exception("Scene must be clean and stopped");
  var saved="/Users/vladislavpanin/Documents/Codex/2026-09-20/localhost-3001-aspid-fasttools-ru-docs/work/other-samples/original-scene.txt";
  if(!File.Exists(saved))File.WriteAllText(saved,scene.path);
  var assembly=AppDomain.CurrentDomain.GetAssemblies().First(a=>a.GetName().Name=="Aspid.FastTools.Editor");
  var settingsType=assembly.GetType("Aspid.FastTools.Editors.Internal.SampleThemeSettings");
  var settings=(UnityEngine.Object)settingsType.BaseType.GetProperty("instance",Flags).GetValue(null);
  var serialized=new SerializedObject(settings);
  foreach(var pair in new[]{("Cyan","97C7D8"),("Grid","B4BDC8"),("Porcelain","CFD9DF")}) {
   ColorUtility.TryParseHtmlString("#"+pair.Item2,out var color);
   serialized.FindProperty("_light."+pair.Item1).colorValue=color;
  }
  serialized.FindProperty("_light.AnimatedColorLift").floatValue=0.45f;
  serialized.ApplyModifiedPropertiesWithoutUndo();settingsType.GetMethod("Persist",Flags).Invoke(settings,null);
  var preview=assembly.GetType("Aspid.FastTools.Editors.Internal.SampleThemePreview");
  var mode=preview.GetMethod("SetMode",Flags);
  mode.Invoke(null,new[]{Enum.ToObject(mode.GetParameters()[0].ParameterType,0)});
  scene=EditorSceneManager.OpenScene("Assets/Samples/Aspid.FastTools/1.0.0-rc.8/"+sample+"/Scenes/"+sample+".unity");
  mode.Invoke(null,new[]{Enum.ToObject(mode.GetParameters()[0].ParameterType,2)});
  return new{sample,scene=scene.path,dirty=scene.isDirty};
 }
}
