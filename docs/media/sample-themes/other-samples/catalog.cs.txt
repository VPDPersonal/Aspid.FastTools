// Ability Catalog states for the EditorTools demo, grabbed from a hidden batch-mode Editor with the asp-unity-capture
// skill: each block below is one `eval.sh <project> <block.cs> Aspid.FastTools.Samples.EditorTools.Editors.AbilityCatalogWindow`
// step (the skill prepends win, root, priv, all, byText, pseudo, repaint, rebuild, grab). Run it in a scratch worktree
// (`editor.sh worktree HEAD <dir>`): the real button writes Fireball.asset and Undo restores it.
//
// Setup, once:
//   1. copy the pristine sample `Packages/tech.aspid.fasttools/Samples~/EditorTools` to
//      `Assets/Samples/Aspid.FastTools/<version>/EditorTools` of the worktree (Arcane Ward, Fireball, Healing Word, Sprint),
//      then a step with `AssetDatabase.Refresh(ImportAssetOptions.ForceSynchronousImport)` and wait for the compile;
//   2. `editor.sh window <project> Aspid.FastTools.Samples.EditorTools.Editors.AbilityCatalogWindow`.
// The window uses the FastTools theme picked under Appearance; leave it on Default. Light comes from the editor skin:
// dark first, then switch with `InternalEditorUtility.SwitchSkinAndRepaintAllViews()` and restore the shared UserSkin key
// as the skill's "Light skin" section says. Per skin run "setup", "idle", "hover", "hover-grab", "click", "after",
// "undo", "undone" into catalog/<theme>/. The undone grab must equal idle.png byte for byte; the specs reuse idle.png for
// it, so undone.png is not kept.

// --- head: prepended to every step
var list = (UnityEngine.UIElements.ListView)priv(win, "_list");
var reset = (UnityEngine.UIElements.Button)byText("Reset to defaults");
System.Func<string> values = () =>
{
    var so = new UnityEditor.SerializedObject((UnityEngine.Object)list.selectedItem);
    var row = all(root).Find(e => e.ClassListContains("unity-collection-view__item--selected"));
    return so.targetObject.name + " " + so.FindProperty("_manaCost").intValue + " MP / " + so.FindProperty("_cooldown").floatValue
        + " s; row '" + ((UnityEngine.UIElements.Label)UnityEngine.UIElements.UQueryExtensions.Q(row, "abilityStats")).text
        + "'; reset " + (reset == null ? "missing" : reset.resolvedStyle.display.ToString());
};

// --- setup: 1000 × 563 pt, empty search, Fireball selected, no hover, no focus, empty Undo history
var p = win.position; win.position = new UnityEngine.Rect(p.x, p.y, 1000, 563);
rebuild();
var search = (UnityEngine.UIElements.TextField)priv(win, "_search"); search.value = "";
list = (UnityEngine.UIElements.ListView)priv(win, "_list");
var filtered = (System.Collections.IList)priv(win, "_filtered");
for (var i = 0; i < filtered.Count; i++) if (((UnityEngine.Object)filtered[i]).name == "Fireball") list.SetSelection(i);
foreach (var e in all(root)) pseudo(e, "Hover", false);
var focused = root.panel.focusController.focusedElement as UnityEngine.UIElements.VisualElement; if (focused != null) focused.Blur();
UnityEditor.Undo.ClearAll();
repaint();
return "pro=" + UnityEditor.EditorGUIUtility.isProSkin;

// --- idle: 30 MP / 4 s, Reset to defaults visible at (428, 397.5, 147, 32) pt
repaint(); repaint();
reset = (UnityEngine.UIElements.Button)byText("Reset to defaults");
return grab("<out>/idle.png") + " | " + values() + " | reset rect " + reset.worldBound;

// --- hover
pseudo(reset, "Hover", true); repaint();
return "hover on";

// --- hover-grab
repaint(); repaint();
return grab("<out>/hover.png") + " | " + values();

// --- click: the button's own handler (SetFloat + SetIntAndApply of the AbilityConfig defaults)
pseudo(reset, "Hover", false);
((System.Delegate)priv(reset.clickable, "clicked")).DynamicInvoke();
return "clicked: " + values();

// --- after: 10 MP / 1 s, the row reads "10 MP · 1 s" and the button is hidden (both follow one editor tick later)
repaint(); repaint();
return grab("<out>/after.png") + " | " + values();

// --- undo
UnityEditor.Undo.PerformUndo();
return "undo: " + values();

// --- undone: 30 MP / 4 s and the button again; compare with idle.png
repaint(); repaint();
return grab("<out>/undone.png") + " | " + values();
