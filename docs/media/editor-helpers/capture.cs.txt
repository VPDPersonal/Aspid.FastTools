var preview = UnityEditor.SceneManagement.EditorSceneManager.NewPreviewScene();
var go = new UnityEngine.GameObject("Documentation preview");
UnityEngine.SceneManagement.SceneManager.MoveGameObjectToScene(go, preview);
var first = go.AddComponent<UnityEngine.AudioSource>();
var second = go.AddComponent<UnityEngine.AudioSource>();
var box = go.AddComponent<UnityEngine.BoxCollider>();
var window = UnityEngine.ScriptableObject.CreateInstance<UnityEditor.EditorWindow>();
window.titleContent = new UnityEngine.GUIContent("Editor Helpers — API results");
window.position = new UnityEngine.Rect(180, 140, 760, 260);
var root = window.rootVisualElement;
root.style.backgroundColor = new UnityEngine.Color(0.12f, 0.12f, 0.12f);
root.style.paddingLeft = 24;
root.style.paddingRight = 24;
root.style.paddingTop = 20;
root.style.paddingBottom = 20;
var title = new UnityEngine.UIElements.Label("Readable component names");
title.style.fontSize = 24;
title.style.unityFontStyleAndWeight = UnityEngine.FontStyle.Bold;
title.style.marginBottom = 16;
root.Add(title);
void Row(string method, string result) {
    var row = new UnityEngine.UIElements.VisualElement();
    row.style.flexDirection = UnityEngine.UIElements.FlexDirection.Row;
    row.style.paddingTop = 10;
    row.style.paddingBottom = 10;
    var left = new UnityEngine.UIElements.Label(method);
    left.style.fontSize = 20;
    left.style.width = 405;
    var right = new UnityEngine.UIElements.Label(result);
    right.style.fontSize = 22;
    right.style.color = new UnityEngine.Color(0.65f, 0.86f, 0.5f);
    row.Add(left);
    row.Add(right);
    root.Add(row);
}
Row("audio.GetScriptName()", Aspid.FastTools.Editors.EditorExtensions.GetScriptName(first));
Row("secondAudio.GetScriptNameWithIndex()", Aspid.FastTools.Editors.EditorExtensions.GetScriptNameWithIndex(second));
Row("collider.GetScriptName()", Aspid.FastTools.Editors.EditorExtensions.GetScriptName(box));
UnityEditor.SceneManagement.EditorSceneManager.ClosePreviewScene(preview);
window.ShowUtility();
window.Focus();
return window.GetInstanceID();
