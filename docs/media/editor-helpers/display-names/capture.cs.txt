using UnityEngine;
using UnityEditor;
using UnityEngine.UIElements;
using Aspid.FastTools.Editors;

namespace DocumentationCapture {
    public sealed class AbilityConfig : MonoBehaviour { }

    public static class DisplayNames {
        public static object Main() {
            var preview = UnityEditor.SceneManagement.EditorSceneManager.NewPreviewScene();
            var go = new GameObject("Documentation preview");
            UnityEngine.SceneManagement.SceneManager.MoveGameObjectToScene(go, preview);
            var first = go.AddComponent<AbilityConfig>();
            var second = go.AddComponent<AbilityConfig>();
            var values = new[] { first.GetDisplayName(), first.GetDisplayNameWithIndex(), second.GetDisplayNameWithIndex() };
            UnityEditor.SceneManagement.EditorSceneManager.ClosePreviewScene(preview);
            var window = ScriptableObject.CreateInstance<EditorWindow>();
            window.titleContent = new GUIContent("Editor Helpers — display names");
            window.minSize = window.maxSize = new Vector2(820, 290);
            window.position = new Rect(140, 100, 820, 290);
            SessionState.SetInt("Docs.DisplayNames.Window", window.GetInstanceID());
            Paint(window, false, values);
            window.ShowUtility();
            window.Focus();
            return new { values, window = window.GetInstanceID(), scene = UnityEditor.SceneManagement.EditorSceneManager.GetActiveScene().path };
        }
        public static void Paint(EditorWindow window, bool light, string[] values) {
            var root = window.rootVisualElement;
            root.Clear();
            root.style.backgroundColor = light ? (Color)new Color32(239,240,243,255) : (Color)new Color32(40,42,46,255);
            root.style.color = light ? (Color)new Color32(42,46,50,255) : (Color)new Color32(223,227,232,255);
            root.style.paddingLeft = root.style.paddingRight = 26;
            root.style.paddingTop = 22;
            root.style.paddingBottom = 22;
            void Heading(string text) {
                var label = new Label(text);
                label.style.fontSize = 21;
                label.style.unityFontStyleAndWeight = FontStyle.Bold;
                label.style.marginBottom = 9;
                root.Add(label);
            }
            void Row(string call, string result) {
                var row = new VisualElement();
                row.style.flexDirection = FlexDirection.Row;
                row.style.height = 44;
                row.style.alignItems = Align.Center;
                var method = new Label(call);
                method.style.width = 472;
                method.style.fontSize = 18;
                var value = new Label(result);
                value.style.fontSize = 20;
                value.style.color = light ? (Color)new Color32(50,103,60,255) : (Color)new Color32(169,211,145,255);
                row.Add(method); row.Add(value); root.Add(row);
            }
            Heading("Readable labels");
            Row("abilityConfig.GetDisplayName()", values[0]);
            var gap = new VisualElement(); gap.style.height = 21; root.Add(gap);
            Heading("Two AbilityConfig components on one GameObject");
            Row("first.GetDisplayNameWithIndex()", values[1]);
            Row("second.GetDisplayNameWithIndex()", values[2]);
            window.Repaint();
        }
    }
}
